2Google Sheets — voor dagboekdata

Eénmalig door Bert in te stellen (alleen via PC/laptop):

A Ga naar sheets.google.com → maak een nieuw spreadsheet aan → noem het Colombia Dagboek.
Hernoem het eerste tabblad naar Data.

B In het spreadsheet: Uitbreidingen → Apps Script → verwijder alle bestaande code → plak dit: 

const SECRET = 'd-Vs_5afzPjCjSgppSCMAmGocAlMqj-XUa-L5QPKny4';

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    if (body.token !== SECRET)
      return maakAntwoord('403 Forbidden');

    const ss  = SpreadsheetApp.getActiveSpreadsheet();
    const blad = ss.getSheets()[0];

    if (body.actie === 'lees') {
      const data = blad.getRange('A1').getValue() || '{}';
      return ContentService
        .createTextOutput(data)
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (body.actie === 'schrijf') {
      blad.getRange('A1').setValue(body.data);
      const backup = ss.getSheetByName('Backup')
                  || ss.insertSheet('Backup');
      backup.appendRow([new Date(), body.data]);
      return maakAntwoord('OK');
    }

    return maakAntwoord('Onbekende actie');

  } catch(err) {
    return maakAntwoord('Fout: ' + err.toString());
  }
}

function doGet(e) {
  return maakAntwoord('403 Forbidden');
}

function maakAntwoord(tekst) {
  return ContentService.createTextOutput(tekst);
}

C Klik Implementeren → Nieuwe implementatie:
• Type: Web-app
• Uitvoeren als: Ik (jouw Google account)
• Toegang: Iedereen (geen Google login nodig voor de app zelf — de token zorgt voor de beveiliging)
→ Klik Implementeren → kopieer de Web App URL (begint met https://script.google.com/macros/s/...)

D Plak de URL hieronder. Het token staat al ingebakken in de code — je hoeft dat niet apart in te voeren als je de code hierboven ongewijzigd gebruikt.
